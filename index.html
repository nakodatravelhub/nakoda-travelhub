function downloadCSV(filename, rows){ let csv=""; rows.forEach(r=>csv+=r.join(",")+"\n"); const blob=new Blob([csv],{type:"text/csv"}); const url=URL.createObjectURL(blob); const a=document.createElement("a"); a.href=url; a.download=filename; a.click(); URL.revokeObjectURL(url); }
